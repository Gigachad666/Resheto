cmake_minimum_required(VERSION 3.21)
project(ReshetoNetworkConsumer LANGUAGES CXX)

set(RESHETO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Resheto source directory")
set(RESHETO_BUILD_GUI OFF CACHE BOOL "" FORCE)
set(RESHETO_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(RESHETO_BUILD_NETWORK ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${RESHETO_ROOT}" resheto)

add_executable(resheto-network-consumer main.cpp)
target_compile_features(resheto-network-consumer PRIVATE cxx_std_20)
target_compile_definitions(resheto-network-consumer PRIVATE RESHETO_ROOT="${RESHETO_ROOT}")
target_link_libraries(resheto-network-consumer PRIVATE Resheto::Network)
if(MSVC)
  target_compile_options(resheto-network-consumer PRIVATE /utf-8 /W4 /permissive-)
endif()
if(WIN32)
  target_link_libraries(resheto-network-consumer PRIVATE psapi)
endif()
