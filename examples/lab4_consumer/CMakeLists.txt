cmake_minimum_required(VERSION 3.21)
project(ReshetoLab4Consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(RESHETO_BUILD_GUI OFF CACHE BOOL "" FORCE)
set(RESHETO_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
if(NOT RESHETO_SOURCE)
  message(FATAL_ERROR "Задайте RESHETO_SOURCE — каталог исходников Решето")
endif()
add_subdirectory("${RESHETO_SOURCE}" resheto)
add_executable(lab4-consumer main.cpp)
target_link_libraries(lab4-consumer PRIVATE Resheto::Core)
if(MSVC)
  target_compile_options(lab4-consumer PRIVATE /utf-8 /W4 /permissive-)
endif()
