cmake_minimum_required(VERSION 3.21)
project(Resheto VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
option(RESHETO_BUILD_GUI "Собирать Qt Widgets GUI" ON)
option(RESHETO_BUILD_TOOLS "Собирать CLI и средства подготовки" ON)
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Sql)
if(RESHETO_BUILD_GUI)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
endif()
if(BUILD_TESTING)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
endif()
if(Qt6_VERSION VERSION_GREATER_EQUAL 7)
  message(FATAL_ERROR "Требуется Qt 6")
endif()
find_package(Threads REQUIRED)
add_library(resheto_core
  src/core/model.cpp src/core/database.cpp src/core/signatures.cpp
  src/core/file_access.cpp src/core/scanner.cpp src/core/controller.cpp src/core/search.cpp src/core/import.cpp)
add_library(Resheto::Core ALIAS resheto_core)
target_include_directories(resheto_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>)
target_link_libraries(resheto_core PUBLIC Qt6::Core Qt6::Sql Threads::Threads)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
  target_compile_options(resheto_core PRIVATE /utf-8 /W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  target_compile_options(resheto_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(RESHETO_BUILD_TOOLS)
add_executable(resheto-cli src/cli.cpp)
target_link_libraries(resheto-cli PRIVATE resheto_core)
add_executable(resheto-prepare src/prepare.cpp)
target_link_libraries(resheto-prepare PRIVATE resheto_core)
add_executable(resheto-import src/import_main.cpp)
target_link_libraries(resheto-import PRIVATE Resheto::Core)
add_executable(resheto-benchmark src/benchmark.cpp)
target_link_libraries(resheto-benchmark PRIVATE Resheto::Core)
if(WIN32)
  target_link_libraries(resheto-benchmark PRIVATE psapi)
endif()
endif()
if(RESHETO_BUILD_GUI)
  add_library(resheto_gui src/gui/mainwindow.cpp)
  target_link_libraries(resheto_gui PUBLIC resheto_core Qt6::Widgets)
  add_executable(resheto WIN32 src/gui.cpp)
  target_link_libraries(resheto PRIVATE resheto_gui)
endif()
if(BUILD_TESTING)
  add_executable(resheto-completion-tests tests/completion_tests.cpp)
  target_link_libraries(resheto-completion-tests PRIVATE Resheto::Core Qt6::Test)
  target_compile_definitions(resheto-completion-tests PRIVATE RESHETO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME completion COMMAND resheto-completion-tests -o completion-results.txt,txt -o -,txt)
  set_tests_properties(completion PROPERTIES TIMEOUT 120)
  add_executable(resheto-audit-tests tests/audit_tests.cpp)
  target_link_libraries(resheto-audit-tests PRIVATE resheto_core Qt6::Test)
  add_test(NAME audit COMMAND resheto-audit-tests -o audit-results.txt,txt -o -,txt)
  set_tests_properties(audit PROPERTIES TIMEOUT 120)
  add_executable(resheto-tests tests/core_tests.cpp)
  target_link_libraries(resheto-tests PRIVATE resheto_core Qt6::Test)
  add_test(NAME core COMMAND resheto-tests -o core-results.txt,txt -o -,txt)
  if(RESHETO_BUILD_TOOLS)
  add_executable(resheto-integration tests/integration.cpp)
  target_link_libraries(resheto-integration PRIVATE resheto_core Qt6::Test)
  add_test(NAME cli COMMAND resheto-integration $<TARGET_FILE:resheto-cli> $<TARGET_FILE:resheto-prepare> ${CMAKE_CURRENT_SOURCE_DIR}/data/demo_signatures.json)
  set_tests_properties(cli PROPERTIES TIMEOUT 120)
  if(WIN32)
    find_program(RESHETO_POWERSHELL powershell.exe)
    if(RESHETO_POWERSHELL)
      add_test(NAME scripts-powershell51 COMMAND "${RESHETO_POWERSHELL}" -NoProfile -NonInteractive -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/powershell_scripts.ps1" -Bin "$<TARGET_FILE_DIR:resheto-cli>" -WorkDir "${CMAKE_CURRENT_BINARY_DIR}")
      set_tests_properties(scripts-powershell51 PROPERTIES TIMEOUT 120)
    endif()
  endif()
  endif()
  set_tests_properties(core PROPERTIES TIMEOUT 120)
  if(RESHETO_BUILD_GUI)
    add_executable(resheto-gui-tests tests/gui_tests.cpp)
    target_link_libraries(resheto-gui-tests PRIVATE resheto_gui Qt6::Test)
    add_test(NAME gui COMMAND resheto-gui-tests -platform offscreen -o gui-results.txt,txt -o -,txt)
    set_tests_properties(gui PROPERTIES TIMEOUT 60)
  endif()
endif()
