cmake_minimum_required(VERSION 3.21)
project(Resheto VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
option(RESHETO_BUILD_GUI "Собирать Qt Widgets GUI" ON)
option(RESHETO_BUILD_TOOLS "Собирать CLI и средства подготовки" ON)
option(RESHETO_BUILD_NETWORK "Собирать сетевой модуль" ${RESHETO_BUILD_GUI})
include(CTest)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Sql)
if(RESHETO_BUILD_GUI)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
endif()
if(RESHETO_BUILD_NETWORK)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Network)
endif()
if(BUILD_TESTING)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Test)
endif()
if(Qt6_VERSION VERSION_GREATER_EQUAL 7)
  message(FATAL_ERROR "Требуется Qt 6")
endif()
find_package(Threads REQUIRED)
add_library(resheto_core
  src/core/model.cpp src/core/database.cpp src/core/signatures.cpp
  src/core/file_access.cpp src/core/scanner.cpp src/core/controller.cpp src/core/search.cpp src/core/import.cpp
  src/core/executable.cpp src/core/heuristic.cpp src/core/process.cpp src/core/behavior.cpp
  src/core/file_monitor.cpp src/core/monitor.cpp)
add_library(Resheto::Core ALIAS resheto_core)
target_include_directories(resheto_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>)
target_link_libraries(resheto_core PUBLIC Qt6::Core Qt6::Sql Threads::Threads)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
  target_compile_options(resheto_core PRIVATE /utf-8 /W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  target_compile_options(resheto_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(RESHETO_BUILD_NETWORK)
  add_library(resheto_network
    src/network/capture.cpp src/network/model.cpp src/network/packet.cpp
    src/network/pcap.cpp src/network/rules.cpp src/network/analyzer.cpp
    src/network/store.cpp src/network/runner.cpp src/network/protocol.cpp
    src/network/private_session.cpp src/network/session.cpp src/network/worker.cpp)
  add_library(Resheto::Network ALIAS resheto_network)
  target_include_directories(resheto_network PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>)
  target_link_libraries(resheto_network PUBLIC Resheto::Core Qt6::Network)
  if(WIN32)
    target_include_directories(resheto_network PRIVATE src/network/vendor)
    target_link_libraries(resheto_network PRIVATE ws2_32 advapi32)
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PCAP REQUIRED IMPORTED_TARGET libpcap>=1.10)
    target_link_libraries(resheto_network PRIVATE PkgConfig::PCAP)
  endif()
  if(RESHETO_BUILD_TOOLS OR RESHETO_BUILD_GUI OR BUILD_TESTING)
    add_executable(resheto-network-worker src/network_worker_main.cpp)
    target_link_libraries(resheto-network-worker PRIVATE Resheto::Network)
  endif()
  if(RESHETO_BUILD_TOOLS)
    add_executable(resheto-network src/network_cli.cpp)
    target_link_libraries(resheto-network PRIVATE Resheto::Network)
    add_dependencies(resheto-network resheto-network-worker)
    add_executable(resheto-network-prepare src/network_prepare.cpp)
    target_link_libraries(resheto-network-prepare PRIVATE Resheto::Network)
  endif()
  if(RESHETO_BUILD_TOOLS OR RESHETO_BUILD_GUI)
    add_executable(resheto-network-sample src/network_sample.cpp)
    target_link_libraries(resheto-network-sample PRIVATE Resheto::Core Qt6::Network)
  endif()
endif()
if(RESHETO_BUILD_TOOLS OR BUILD_TESTING)
  add_executable(resheto-safe-helper tests/helpers/safe_process.cpp)
  target_link_libraries(resheto-safe-helper PRIVATE Qt6::Core)
endif()
if(RESHETO_BUILD_TOOLS)
add_executable(resheto-cli src/cli.cpp)
target_link_libraries(resheto-cli PRIVATE resheto_core)
add_executable(resheto-monitor src/monitor_cli.cpp)
target_link_libraries(resheto-monitor PRIVATE Resheto::Core)
add_executable(resheto-lab4-prepare src/prepare_lab4.cpp)
target_link_libraries(resheto-lab4-prepare PRIVATE Resheto::Core)
add_executable(resheto-prepare src/prepare.cpp)
target_link_libraries(resheto-prepare PRIVATE resheto_core)
add_executable(resheto-import src/import_main.cpp)
target_link_libraries(resheto-import PRIVATE Resheto::Core)
add_executable(resheto-benchmark src/benchmark.cpp)
target_link_libraries(resheto-benchmark PRIVATE Resheto::Core)
if(WIN32)
  target_link_libraries(resheto-benchmark PRIVATE psapi)
endif()
endif()
if(RESHETO_BUILD_GUI)
  add_library(resheto_gui src/gui/mainwindow.cpp src/gui/startup.cpp)
  target_link_libraries(resheto_gui PUBLIC resheto_core Qt6::Widgets)
  if(RESHETO_BUILD_NETWORK)
    target_sources(resheto_gui PRIVATE src/gui/networkpanel.cpp src/gui/networkpanel.h)
    target_link_libraries(resheto_gui PUBLIC Resheto::Network)
    target_compile_definitions(resheto_gui PUBLIC RESHETO_WITH_NETWORK=1)
    add_dependencies(resheto_gui resheto-network-worker resheto-network-sample)
  endif()
  add_executable(resheto WIN32 src/gui.cpp)
  target_link_libraries(resheto PRIVATE resheto_gui)
endif()
if(BUILD_TESTING)
  if(RESHETO_BUILD_NETWORK)
    add_executable(resheto-network-tests tests/network_tests.cpp)
    target_link_libraries(resheto-network-tests PRIVATE Resheto::Network Qt6::Test)
    target_compile_definitions(resheto-network-tests PRIVATE RESHETO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME network COMMAND resheto-network-tests -o network-results.txt,txt -o -,txt)
    set_tests_properties(network PROPERTIES TIMEOUT 90)
    add_executable(resheto-network-runtime-tests tests/network_runtime_tests.cpp)
    target_link_libraries(resheto-network-runtime-tests PRIVATE Resheto::Network Qt6::Test)
    target_compile_definitions(resheto-network-runtime-tests PRIVATE RESHETO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" RESHETO_NETWORK_WORKER="$<TARGET_FILE:resheto-network-worker>" RESHETO_NETWORK_TEST_SELF="$<TARGET_FILE:resheto-network-runtime-tests>")
    add_dependencies(resheto-network-runtime-tests resheto-network-worker)
    add_test(NAME network-runtime COMMAND resheto-network-runtime-tests -o network-runtime-results.txt,txt -o -,txt)
    set_tests_properties(network-runtime PROPERTIES TIMEOUT 120)
  endif()
  add_executable(resheto-monitor-tests tests/monitor_tests.cpp)
  target_link_libraries(resheto-monitor-tests PRIVATE Resheto::Core Qt6::Test)
  target_compile_definitions(resheto-monitor-tests PRIVATE RESHETO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" RESHETO_HELPER_PATH="$<TARGET_FILE:resheto-safe-helper>")
  add_dependencies(resheto-monitor-tests resheto-safe-helper)
  add_test(NAME monitor COMMAND resheto-monitor-tests -o monitor-results.txt,txt -o -,txt)
  set_tests_properties(monitor PROPERTIES TIMEOUT 60)
  add_executable(resheto-lab4-tests tests/lab4_tests.cpp)
  target_link_libraries(resheto-lab4-tests PRIVATE Resheto::Core Qt6::Test)
  target_compile_definitions(resheto-lab4-tests PRIVATE RESHETO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME lab4 COMMAND resheto-lab4-tests -o lab4-results.txt,txt -o -,txt)
  set_tests_properties(lab4 PROPERTIES TIMEOUT 90)
  add_executable(resheto-completion-tests tests/completion_tests.cpp)
  target_link_libraries(resheto-completion-tests PRIVATE Resheto::Core Qt6::Test)
  target_compile_definitions(resheto-completion-tests PRIVATE RESHETO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME completion COMMAND resheto-completion-tests -o completion-results.txt,txt -o -,txt)
  set_tests_properties(completion PROPERTIES TIMEOUT 120)
  add_executable(resheto-audit-tests tests/audit_tests.cpp)
  target_link_libraries(resheto-audit-tests PRIVATE resheto_core Qt6::Test)
  add_test(NAME audit COMMAND resheto-audit-tests -o audit-results.txt,txt -o -,txt)
  set_tests_properties(audit PROPERTIES TIMEOUT 120)
  add_executable(resheto-tests tests/core_tests.cpp)
  target_link_libraries(resheto-tests PRIVATE resheto_core Qt6::Test)
  add_test(NAME core COMMAND resheto-tests -o core-results.txt,txt -o -,txt)
  if(RESHETO_BUILD_TOOLS)
  add_executable(resheto-integration tests/integration.cpp)
  target_link_libraries(resheto-integration PRIVATE resheto_core Qt6::Test)
  add_test(NAME cli COMMAND resheto-integration $<TARGET_FILE:resheto-cli> $<TARGET_FILE:resheto-prepare> ${CMAKE_CURRENT_SOURCE_DIR}/data/demo_signatures.json)
  set_tests_properties(cli PROPERTIES TIMEOUT 120)
  if(WIN32)
    find_program(RESHETO_POWERSHELL powershell.exe)
    if(RESHETO_POWERSHELL)
      add_test(NAME scripts-powershell51 COMMAND "${RESHETO_POWERSHELL}" -NoProfile -NonInteractive -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/powershell_scripts.ps1" -Bin "$<TARGET_FILE_DIR:resheto-cli>" -WorkDir "${CMAKE_CURRENT_BINARY_DIR}")
      set_tests_properties(scripts-powershell51 PROPERTIES TIMEOUT 120)
    endif()
  endif()
  endif()
  set_tests_properties(core PROPERTIES TIMEOUT 120)
  if(RESHETO_BUILD_GUI)
    if(RESHETO_BUILD_NETWORK)
      add_executable(resheto-network-gui-tests tests/network_gui_tests.cpp)
      target_link_libraries(resheto-network-gui-tests PRIVATE resheto_gui Qt6::Test)
      target_compile_definitions(resheto-network-gui-tests PRIVATE RESHETO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
      add_dependencies(resheto-network-gui-tests resheto-network-worker)
      add_test(NAME network-gui COMMAND resheto-network-gui-tests -platform offscreen -o network-gui-results.txt,txt -o -,txt preservesHistoryAndFindsDetectionBeyondFirstPage compactDetailsAndSeparateResultViews networkCancelLeavesManualAndWatchRunning)
      set_tests_properties(network-gui PROPERTIES TIMEOUT 90)
    endif()
    add_executable(resheto-gui-tests tests/gui_tests.cpp)
    target_link_libraries(resheto-gui-tests PRIVATE resheto_gui Qt6::Test)
    target_compile_definitions(resheto-gui-tests PRIVATE RESHETO_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}" RESHETO_HELPER_PATH="$<TARGET_FILE:resheto-safe-helper>")
    add_dependencies(resheto-gui-tests resheto-safe-helper)
    add_test(NAME gui COMMAND resheto-gui-tests -platform offscreen -o gui-results.txt,txt -o -,txt)
    # Общий бюджет всей GUI-группы, включая 2101 SQLite commit, а не одного сценария.
    set_tests_properties(gui PROPERTIES TIMEOUT 180)
  endif()
endif()
